# ZeDUI/tools/qa/rt_coverage_plumbing.txt  (Nov 18 2026, QA): ZeD files that are not features themselves, so no row in
# FEATURE_MAP.md owns them: drawing parts, art data, helpers. rt_coverage.ps1 fails on any ZeD file that is neither
# owned by a feature row nor listed here. Add a file here only when it really draws or does nothing a player can
# reach on its own; a new feature gets a row in FEATURE_MAP.md instead. One path per line, then # the reason.

AtomExtensions.lua              # more atoms

PhysPx.lua                      # pixel art data for physical props
DashPx.lua                      # the dashboards' pixel art data (the dash rows test the dash)
MechFigs.lua                    # the Vehicle tab's part figures (art data)
RadioPlates.lua                 # radio face art data (the radio rows test the faces)
WatchPlates.lua                 # watch face art data (the watch rows test the watches)

ZeD_BodyFigure.lua              # the Health tab's body figure drawing

RecipeIndex.lua                 # the recipe lookup the crafting tabs read
ZeD_RadioOps.lua                # radio power and tuning helpers the radio faces call
Overrides/MakeUp.lua            # the game's make-up hook into ZeD_MakeUp.lua (owned by the make-up rows)
